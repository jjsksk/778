<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Fullscreen Canvas</title>
    <style>
        body, html {
            margin: 0;
            padding: 0;
            width: 100%;
            height: 100%;
            background-color: #dde5b6;
        }
        canvas {
            display: block;
        }
    </style>
</head>
<body>
    <canvas id="fullscreenCanvas"></canvas>
    <script>
        const canvas = document.getElementById('fullscreenCanvas');
        const context = canvas.getContext('2d');

        function resizeCanvas() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;

            // 清除畫布
            context.clearRect(0, 0, canvas.width, canvas.height);

            // 設定影像
            const image = new Image();
            image.src = 'path/to/your/image.jpg'; // 替換為你的影像路徑
            image.onload = () => {
                const imageWidth = canvas.width * 0.8;
                const imageHeight = canvas.height * 0.8;
                const x = (canvas.width - imageWidth) / 2;
                const y = (canvas.height - imageHeight) / 2;

                // 繪製影像
                context.drawImage(image, x, y, imageWidth, imageHeight);
            };
        }

        window.addEventListener('resize', resizeCanvas);
        resizeCanvas();
    </script>
</body>
</html>
